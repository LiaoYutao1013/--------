# 🎭 非线性滤波与形态学操作

## 概述

非线性滤波相对于[[../Image_Filtering_&_Enhancement/Linear Filters|线性滤波]]，不满足叠加原理。它们能够更好地保护边缘、去除噪声、增强细节。主要包括：
- 中值滤波（Median Filter）
- 双边滤波（Bilateral Filter）
- 形态学操作（Morphological Operations）

**学习时间**：2-3 小时  
**先修课程**：[[../Image_Filtering_&_Enhancement/Linear Filters|线性滤波]]、[[../Image_Representation_&_Color_Spaces/颜色空间与表示|颜色空间]]

## 1. 中值滤波（Median Filter）

### 1.1 原理

对核内所有像素的值进行**排序**，取中间值作为中心像素的新值：

$$f'(x, y) = \text{median}\{f(i, j) : (i,j) \in W(x,y)\}$$

其中 $W(x,y)$ 是以 $(x,y)$ 为中心的窗口。

**特点**：
- ✅ 对脉冲噪声（椒盐噪声）非常有效
- ✅ 边缘保护能力强
- ✅ 对高斯噪声的处理一般
- ❌ 计算复杂度高于线性滤波
- ❌ 对于较大的核，处理速度较慢

### 1.2 OpenCV 实现

```python
import cv2
import numpy as np

# 基本用法
result = cv2.medianBlur(img, 5)  # 5x5 核

# 对比线性高斯模糊
gaussian = cv2.GaussianBlur(img, (5, 5), 1.0)

# 应用场景：椒盐噪声去除
def test_median_filter():
    # 创建测试图像
    img = np.random.randint(0, 256, (100, 100, 3), dtype=np.uint8)
    
    # 添加椒盐噪声
    noise = np.copy(img)
    for _ in range(1000):
        x, y = np.random.randint(0, 100, 2)
        noise[x, y] = [0, 0, 0] if np.random.rand() > 0.5 else [255, 255, 255]
    
    # 应用中值滤波
    denoised = cv2.medianBlur(noise, 5)
    
    # 对比高斯滤波
    gaussian = cv2.GaussianBlur(noise, (5, 5), 1.0)
    
    return denoised, gaussian
```

def median_filter_multichannel(img, kernel_size=5):
    """对每个通道分别进行中值滤波"""
    if len(img.shape) == 3:
        channels = cv2.split(img)
        filtered_channels = [cv2.medianBlur(ch, kernel_size) for ch in channels]
        return cv2.merge(filtered_channels)
    else:
        return cv2.medianBlur(img, kernel_size)

# 快速中值滤波（OpenCV 4.4+）

```python
import time

def benchmark_median_filter():
    """对比不同实现方法的速度"""
    img = np.random.randint(0, 256, (1000, 1000, 3), dtype=np.uint8)
    
    # 方法 1：OpenCV medianBlur (最快)
    start = time.time()
    for _ in range(10):
        result1 = cv2.medianBlur(img, 5)
    time1 = time.time() - start
    
    # 方法 2：多次应用小核 (权衡)
    start = time.time()
    for _ in range(10):
        result2 = cv2.medianBlur(img, 3)
        result2 = cv2.medianBlur(result2, 3)
    time2 = time.time() - start
    
    print(f"Single 5x5 kernel: {time1:.4f}s")
    print(f"Two 3x3 kernels: {time2:.4f}s")
    
    return result1, result2
```

## 2. 双边滤波（Bilateral Filter）

### 2.1 原理

结合**空间距离**和**像素值相似性**的加权平均：

$$f'(x, y) = \frac{\sum_{(i,j) \in W} f(i,j) \cdot w(i,j,x,y)}{\sum_{(i,j) \in W} w(i,j,x,y)}$$

权重函数：
$$w(i,j,x,y) = \exp\left(-\frac{(i-x)^2 + (j-y)^2}{2\sigma_s^2} - \frac{(f(i,j) - f(x,y))^2}{2\sigma_r^2}\right)$$

其中：
- $\sigma_s$：空间方差（模糊程度）
- $\sigma_r$：范围方差（相似性阈值）

**特点**：
- ✅ 保护边缘非常好（边缘保持滤波）
- ✅ 适合去除高斯噪声
- ❌ 计算复杂度最高
- ❌ 处理速度慢（尤其是大核）

### 2.2 基本用法

```python
# 基本双边滤波
result = cv2.bilateralFilter(img, 
                             d=9,              # 直径
                             sigmaColor=75,    # 颜色相似性
                             sigmaSpace=75)    # 空间相似性

# 进阶：自定义参数
def bilateral_denoise(img, iterations=1):
    """迭代双边滤波用于更强的去噪"""
    result = img.copy()
    for _ in range(iterations):
        result = cv2.bilateralFilter(result, d=9, 
                                    sigmaColor=75,
                                    sigmaSpace=75)
    return result

# 应用：皮肤平滑
def smooth_skin(img):
    """美颜效果：皮肤平滑"""
    # 双边滤波：保持边缘，平滑面部
    smoothed = cv2.bilateralFilter(img, d=9, 
                                  sigmaColor=100,
                                  sigmaSpace=100)
    
    # 混合：保留原始细节的一部分
    result = cv2.addWeighted(img, 0.3, smoothed, 0.7, 0)
    return result
```

## 3. 形态学操作（Morphological Operations）

### 3.1 基本概念

形态学操作基于**结构元素**（Structuring Element）对二值或灰度图像进行处理。

#### 三种基本操作

**腐蚀（Erosion）**：取核内的**最小值**
$$\text{Erosion}(f) = \min\{f(i,j) : (i,j) \in S\}$$

```python
eroded = cv2.erode(img, kernel, iterations=1)
```

**膨胀（Dilation）**：取核内的**最大值**
$$\text{Dilation}(f) = \max\{f(i,j) : (i,j) \in S\}$$

```python
dilated = cv2.dilate(img, kernel, iterations=1)
```

**复合操作**：
- 开运算（Opening）= 腐蚀 + 膨胀（去除小对象）
- 闭运算（Closing）= 膨胀 + 腐蚀（填充小孔洞）

### 3.2 结构元素设计

### 3.3 基本形态学操作

```python
def morphological_operations(img):
    """演示各种形态学操作"""
    
    # 创建测试二值图像
    binary = cv2.threshold(cv2.cvtColor(img, cv2.COLOR_BGR2GRAY), 
                          127, 255, cv2.THRESH_BINARY)[1]
    
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    
    # 腐蚀
    eroded = cv2.erode(binary, kernel, iterations=1)
    
    # 膨胀
    dilated = cv2.dilate(binary, kernel, iterations=1)
    
    # 开运算（去噪）
    opened = cv2.morphologyEx(binary, cv2.MORPH_OPEN, kernel)
    
    # 闭运算（填充）
    closed = cv2.morphologyEx(binary, cv2.MORPH_CLOSE, kernel)
    
    # 梯度（边界检测）
    gradient = cv2.morphologyEx(binary, cv2.MORPH_GRADIENT, kernel)
    
    # 顶帽（突出小对象）
    tophat = cv2.morphologyEx(binary, cv2.MORPH_TOPHAT, kernel)
    
    # 黑帽（突出孔洞）
    blackhat = cv2.morphologyEx(binary, cv2.MORPH_BLACKHAT, kernel)
    
    return {
        'original': binary,
        'eroded': eroded,
        'dilated': dilated,
        'opened': opened,
        'closed': closed,
        'gradient': gradient,
        'tophat': tophat,
        'blackhat': blackhat
    }
```

### 3.4 复杂形态学操作

```python
def advanced_morphological_ops(img):
    """高级形态学应用"""
    
    binary = cv2.threshold(cv2.cvtColor(img, cv2.COLOR_BGR2GRAY), 
                          127, 255, cv2.THRESH_BINARY)[1]
    
    # 多次迭代腐蚀/膨胀
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3))
    
    # 强力腐蚀（多次迭代）
    strongly_eroded = cv2.erode(binary, kernel, iterations=5)
    
    # 强力膨胀
    strongly_dilated = cv2.dilate(binary, kernel, iterations=5)
    
    # 连通分量去除
    def remove_small_components(img, min_size=100):
        """移除小于 min_size 的连通分量"""
        nlabels, labels, stats, centroids = cv2.connectedComponentsWithStats(img)
        
        result = np.zeros_like(img)
        for i in range(1, nlabels):
            if stats[i, cv2.CC_STAT_AREA] >= min_size:
                result[labels == i] = 255
        
        return result
    
    cleaned = remove_small_components(binary, min_size=50)
    
    return cleaned
```

## 4. 滤波对比与选择

### 4.1 不同滤波器的效果对比

```python
def compare_filters(img):
    """对比各种滤波方法"""
    
    # 添加噪声
    noisy = img + np.random.normal(0, 25, img.shape).astype(np.uint8)
    
    results = {
        'Original': img,
        'Noisy': noisy,
        'Gaussian': cv2.GaussianBlur(noisy, (5, 5), 1.0),
        'Median': cv2.medianBlur(noisy, 5),
        'Bilateral': cv2.bilateralFilter(noisy, d=9, 
                                        sigmaColor=75, sigmaSpace=75),
        'Morphological_Open': cv2.morphologyEx(
            cv2.threshold(noisy, 127, 255, cv2.THRESH_BINARY)[1],
            cv2.MORPH_OPEN, cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))),
    }
    
    # 计算 PSNR
    def psnr(original, filtered):
        mse = np.mean((original - filtered) ** 2)
        if mse == 0:
            return float('inf')
        max_pixel = 255.0
        return 20 * np.log10(max_pixel / np.sqrt(mse))
    
    print("| 滤波方法 | PSNR | 优点 |")
    print("|---------|------|------|")
    print(f"| 高斯 | {psnr(img, results['Gaussian']):.2f} | 快速 |")
    print(f"| 中值 | {psnr(img, results['Median']):.2f} | 去椒盐 |")
    print(f"| 双边 | {psnr(img, results['Bilateral']):.2f} | 保边缘 |")
    
    return results
```

### 4.2 选择指南

```
噪声类型 → 推荐滤波器

高斯噪声 → 高斯滤波 or 双边滤波
椒盐噪声 → 中值滤波
脉冲噪声 → 中值滤波
  
边缘要求 → 滤波方法

需要保护边缘 → 双边滤波、中值滤波
边缘可以模糊 → 高斯滤波
需要二值化 → 形态学操作

实时性要求 → 滤波方法

实时性强 → 高斯滤波（最快）
实时性中等 → 中值滤波
实时性弱 → 双边滤波（最慢）
```

## 5. 实际应用案例

### 5.1 自适应滤波系统

```python
class AdaptiveFilteringPipeline:
    """自适应滤波管道"""
    
    def __init__(self, img):
        self.img = img
        self.gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    
    def estimate_noise_level(self):
        """估计图像噪声水平"""
        # 使用 Laplacian 方差作为清晰度指标
        laplacian_var = cv2.Laplacian(self.gray, cv2.CV_64F).var()
        return laplacian_var
    
    def adaptive_filter(self):
        """根据噪声水平自适应选择滤波方法"""
        noise_level = self.estimate_noise_level()
        
        if noise_level > 100:  # 高噪声
            # 使用更激进的滤波
            result = cv2.bilateralFilter(self.img, d=9, 
                                        sigmaColor=100,
                                        sigmaSpace=100)
        elif noise_level > 50:  # 中等噪声
            # 使用中值滤波
            result = cv2.medianBlur(self.img, 5)
        else:  # 低噪声
            # 使用高斯滤波
            result = cv2.GaussianBlur(self.img, (3, 3), 1.0)
        
        return result
    
    def remove_specific_objects(self, size_threshold):
        """使用形态学去除特定大小的对象"""
        binary = cv2.threshold(self.gray, 127, 255, 
                              cv2.THRESH_BINARY)[1]
        
        kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
        
        # 开运算去除小对象
        opened = cv2.morphologyEx(binary, cv2.MORPH_OPEN, kernel)
        
        # 连通分量分析
        nlabels, labels, stats, centroids = cv2.connectedComponentsWithStats(opened)
        
        result = np.zeros_like(binary)
        for i in range(1, nlabels):
            if stats[i, cv2.CC_STAT_AREA] >= size_threshold:
                result[labels == i] = 255
        
        return result
```

### 5.2 手指检测与跟踪

```python
def detect_fingers(img):
    """使用形态学进行手指检测"""
    
    # 肤色检测
    hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    lower = np.array([0, 10, 60])
    upper = np.array([20, 150, 255])
    mask = cv2.inRange(hsv, lower, upper)
    
    # 形态学处理
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    
    # 闭运算填充手掌内部的孔
    closed = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, kernel, iterations=3)
    
    # 开运算去除噪声
    opened = cv2.morphologyEx(closed, cv2.MORPH_OPEN, kernel)
    
    # 梯度提取边界
    gradient = cv2.morphologyEx(opened, cv2.MORPH_GRADIENT, kernel)
    
    # 查找轮廓（手指）
    contours, _ = cv2.findContours(gradient, cv2.RETR_EXTERNAL, 
                                   cv2.CHAIN_APPROX_SIMPLE)
    
    # 过滤手指（按面积和长宽比）
    fingers = []
    for cnt in contours:
        area = cv2.contourArea(cnt)
        if area > 100:  # 最小面积
            x, y, w, h = cv2.boundingRect(cnt)
            aspect_ratio = float(h) / w if w > 0 else 0
            if 0.3 < aspect_ratio < 3:  # 长条形
                fingers.append(cnt)
    
    return opened, fingers
```

# 1. 避免多次转换格式
def efficient_filtering(img):
    # ❌ 不好：多次转换
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

def optimize_kernel_size(img):
    # 对于大图像，更大的核不一定更好
    if img.shape[0] * img.shape[1] > 2000000:  # > 2MP
        kernel_size = 3  # 较小的核
    else:
        kernel_size = 5

# 3. 使用迭代替代大核
def efficient_erosion():
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3))
    
    # 两个小核的迭代 vs 一个大核
    # 结果相似，但两个小核往往更快
    result = cv2.erode(img, kernel, iterations=2)
